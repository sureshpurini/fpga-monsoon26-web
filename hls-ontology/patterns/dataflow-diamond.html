Dataflow Diamond Pattern

The diamond pattern is a fundamental dataflow topology with fork and join stages.

Architecture

         input
            │
       ┌────▼────┐
       │  funcA  │  (Fork: 1 input → 2 outputs)
       └──┬───┬──┘
       c1 │   │ c2
     ┌────▼┐ ┌▼────┐
     │funcB│ │funcC│  (Parallel paths)
     └────┬┘ └┬────┘
       c3 │   │ c4
       ┌──▼───▼──┐
       │  funcD  │  (Join: 2 inputs → 1 output)
       └────┬────┘
            │
         output

Data Transformation

Each function applies a specific operation:

input[i] → funcA → t = input[i] * 3
                   ├── c1 → funcB → c3 = t + 25
                   └── c2 → funcC → c4 = t * 2
                                    ↓
funcD: output = c3 + c4 * 2 = (t + 25) + (t * 2) * 2
                            = (3*input + 25) + (6*input) * 2
                            = 3*input + 25 + 12*input
                            = 15*input + 25

Implementation 1: Simple FIFOs (Implicit Channels)

Arrays become FIFO channels automatically under #pragma HLS DATAFLOW.

Header

// diamond.h
#define N 100
typedef unsigned char data_t;

// Top function
void diamond(data_t vecIn[N], data_t vecOut[N]);

// Sub functions
void funcA(data_t* in, data_t* out1, data_t* out2);
void funcB(data_t* in, data_t* out);
void funcC(data_t* in, data_t* out);
void funcD(data_t* in1, data_t* in2, data_t* out);

Top Function

#include "diamond.h"

void diamond(data_t vecIn[N], data_t vecOut[N]) {
    data_t c1[N], c2[N], c3[N], c4[N];
#pragma HLS DATAFLOW
    funcA(vecIn, c1, c2);
    funcB(c1, c3);
    funcC(c2, c4);
    funcD(c3, c4, vecOut);
}

funcA - Fork (1 input → 2 outputs)

void funcA(data_t* in, data_t* out1, data_t* out2) {
#pragma HLS INLINE OFF
Loop0:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE
        data_t t = in[i] * 3;  // Multiply by 3
        out1[i] = t;           // Send to path 1 (funcB)
        out2[i] = t;           // Send to path 2 (funcC)
    }
}

funcB - Path 1 Processing

void funcB(data_t* in, data_t* out) {
#pragma HLS INLINE OFF
Loop0:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE
        out[i] = in[i] + 25;   // Add 25
    }
}

funcC - Path 2 Processing

void funcC(data_t* in, data_t* out) {
#pragma HLS INLINE OFF
Loop0:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE
        out[i] = in[i] * 2;    // Multiply by 2
    }
}

funcD - Join (2 inputs → 1 output)

void funcD(data_t* in1, data_t* in2, data_t* out) {
#pragma HLS INLINE OFF
Loop0:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE
        out[i] = in1[i] + in2[i] * 2;  // Combine: path1 + path2*2
    }
}

Implementation 2: Explicit hls::stream

More control with explicit stream API, vectorized data types, and memory interface.

Header

// diamond.h
#include <hls_stream.h>
#include <hls_vector.h>

typedef unsigned int data_t;
#define NUM_WORDS 16

// Each vector is 64 bytes (16 x 4 bytes) for efficient memory access
typedef hls::vector<uint32_t, NUM_WORDS> vecOf16Words;

extern "C" {
// Top function (kernel)
void diamond(vecOf16Words* vecIn, vecOf16Words* vecOut, int size);
}

// Sub functions
void load(vecOf16Words* in, hls::stream<vecOf16Words>& out, int size);
void compute_A(hls::stream<vecOf16Words>& in,
               hls::stream<vecOf16Words>& out1,
               hls::stream<vecOf16Words>& out2, int size);
void compute_B(hls::stream<vecOf16Words>& in,
               hls::stream<vecOf16Words>& out, int size);
void compute_C(hls::stream<vecOf16Words>& in,
               hls::stream<vecOf16Words>& out, int size);
void compute_D(hls::stream<vecOf16Words>& in1,
               hls::stream<vecOf16Words>& in2,
               hls::stream<vecOf16Words>& out, int size);
void store(hls::stream<vecOf16Words>& in, vecOf16Words* out, int size);

Top Function (Kernel)

#include "diamond.h"

extern "C" {
void diamond(vecOf16Words* vecIn, vecOf16Words* vecOut, int size) {
#pragma HLS INTERFACE m_axi port=vecIn depth=32
#pragma HLS INTERFACE m_axi port=vecOut depth=32

    hls::stream<vecOf16Words> c0, c1, c2, c3, c4, c5;
    assert(size % 16 == 0);

#pragma HLS DATAFLOW
    load(vecIn, c0, size);           // Memory → Stream
    compute_A(c0, c1, c2, size);     // Fork
    compute_B(c1, c3, size);         // Path 1
    compute_C(c2, c4, size);         // Path 2
    compute_D(c3, c4, c5, size);     // Join
    store(c5, vecOut, size);         // Stream → Memory
}
}

load - Memory to Stream

void load(vecOf16Words* in, hls::stream<vecOf16Words>& out, int size) {
Loop_Ld:
    for (int i = 0; i < size; i++) {
#pragma HLS PERFORMANCE target_ti=32
#pragma HLS LOOP_TRIPCOUNT max=32
        out.write(in[i]);  // Read from DDR, write to stream
    }
}

compute_A - Fork

void compute_A(hls::stream<vecOf16Words>& in,
               hls::stream<vecOf16Words>& out1,
               hls::stream<vecOf16Words>& out2, int size) {
Loop_A:
    for (int i = 0; i < size; i++) {
#pragma HLS PERFORMANCE target_ti=32
#pragma HLS LOOP_TRIPCOUNT max=32
        vecOf16Words t = in.read();
        out1.write(t * 3);  // Vector multiply: all 16 elements * 3
        out2.write(t * 3);
    }
}

compute_B - Path 1

void compute_B(hls::stream<vecOf16Words>& in,
               hls::stream<vecOf16Words>& out, int size) {
Loop_B:
    for (int i = 0; i < size; i++) {
#pragma HLS PERFORMANCE target_ti=32
#pragma HLS LOOP_TRIPCOUNT max=32
        out.write(in.read() + 25);  // Vector add: all 16 elements + 25
    }
}

compute_C - Path 2

void compute_C(hls::stream<vecOf16Words>& in,
               hls::stream<vecOf16Words>& out, int size) {
Loop_C:
    for (int i = 0; i < size; i++) {
#pragma HLS PERFORMANCE target_ti=32
#pragma HLS LOOP_TRIPCOUNT max=32
        out.write(in.read() * 2);   // Vector multiply: all 16 elements * 2
    }
}

compute_D - Join

void compute_D(hls::stream<vecOf16Words>& in1,
               hls::stream<vecOf16Words>& in2,
               hls::stream<vecOf16Words>& out, int size) {
Loop_D:
    for (int i = 0; i < size; i++) {
#pragma HLS PERFORMANCE target_ti=32
#pragma HLS LOOP_TRIPCOUNT max=32
        out.write(in1.read() + in2.read());  // Vector add
    }
}

store - Stream to Memory

void store(hls::stream<vecOf16Words>& in, vecOf16Words* out, int size) {
Loop_St:
    for (int i = 0; i < size; i++) {
#pragma HLS PERFORMANCE target_ti=32
#pragma HLS LOOP_TRIPCOUNT max=32
        out[i] = in.read();  // Read from stream, write to DDR
    }
}

Implementation 3: Stream of Blocks

Block-level streaming with random access within blocks using RAII locks.

Header

// diamond.h
#pragma once
#include "hls_stream.h"
#include "hls_streamofblocks.h"

#define N 100
#define NUM_BLOCKS 10
typedef unsigned char data_t;
typedef data_t block_data_t[NUM_BLOCKS];  // Block of 10 elements

// Top function
void diamond(hls::stream<data_t>& vecIn, hls::stream<data_t>& vecOut);

// Sub functions
void funcA(hls::stream<data_t>& in,
           hls::stream_of_blocks<block_data_t>& out1,
           hls::stream_of_blocks<block_data_t>& out2);
void funcB(hls::stream_of_blocks<block_data_t>& in,
           hls::stream_of_blocks<block_data_t>& out);
void funcC(hls::stream_of_blocks<block_data_t>& in,
           hls::stream_of_blocks<block_data_t>& out);
void funcD(hls::stream_of_blocks<block_data_t>& in1,
           hls::stream_of_blocks<block_data_t>& in2,
           hls::stream<data_t>& out);

Top Function

#include "diamond.h"

void diamond(hls::stream<data_t>& vecIn, hls::stream<data_t>& vecOut) {
#pragma HLS INTERFACE ap_ctrl_chain port=return

    hls::stream_of_blocks<block_data_t> c1, c2, c3, c4;

#pragma HLS DATAFLOW
    funcA(vecIn, c1, c2);
    funcB(c1, c3);
    funcC(c2, c4);
    funcD(c3, c4, vecOut);
}

funcA - Fork (Stream → Blocks)

void funcA(hls::stream<data_t>& in,
           hls::stream_of_blocks<block_data_t>& out1,
           hls::stream_of_blocks<block_data_t>& out2) {
#pragma HLS INLINE OFF

funcA_Loop0:
    for (int i = 0; i < N / NUM_BLOCKS; i++) {  // 10 iterations
#pragma HLS PIPELINE II=10

        // Obtain write locks for both output channels
        // Locks are RAII - released at end of loop iteration
        hls::write_lock<block_data_t> out1L(out1);
        hls::write_lock<block_data_t> out2L(out2);

        // Read 10 elements from stream, write to blocks
    funcA_Loop1:
        for (unsigned int j = 0; j < NUM_BLOCKS; j++) {
            data_t t = in.read() * 3;
            out1L[j] = t;  // Random access OK within block
            out2L[j] = t;
        }
    }  // write_locks released here (RAII)
}

funcB - Path 1 (Block → Block)

void funcB(hls::stream_of_blocks<block_data_t>& in,
           hls::stream_of_blocks<block_data_t>& out) {
#pragma HLS INLINE OFF

funcB_Loop0:
    for (int i = 0; i < N / NUM_BLOCKS; i++) {
#pragma HLS PIPELINE II=5

        // Obtain read lock on input, write lock on output
        hls::read_lock<block_data_t> inL(in);
        hls::write_lock<block_data_t> outL(out);

    funcB_Loop1:
        for (unsigned int j = 0; j < NUM_BLOCKS; j++)
            outL[j] = inL[j] + 25;  // Random access: read any index
    }  // Locks released here
}

funcC - Path 2 (Block → Block)

void funcC(hls::stream_of_blocks<block_data_t>& in,
           hls::stream_of_blocks<block_data_t>& out) {
#pragma HLS INLINE OFF

Loop0:
    for (int i = 0; i < N / NUM_BLOCKS; i++) {
#pragma HLS PIPELINE II=5

        hls::read_lock<block_data_t> inL(in);
        hls::write_lock<block_data_t> outL(out);

        for (unsigned int j = 0; j < NUM_BLOCKS; ++j)
            outL[j] = inL[j] * 2;
    }
}

funcD - Join (Blocks → Stream)

void funcD(hls::stream_of_blocks<block_data_t>& in1,
           hls::stream_of_blocks<block_data_t>& in2,
           hls::stream<data_t>& out) {
#pragma HLS INLINE OFF

Loop0:
    for (int i = 0; i < N / NUM_BLOCKS; i++) {
#pragma HLS PIPELINE II=10

        // Two read locks for the two input paths
        hls::read_lock<block_data_t> in1L(in1);
        hls::read_lock<block_data_t> in2L(in2);

        for (unsigned int j = 0; j < NUM_BLOCKS; ++j)
            out.write(in1L[j] + in2L[j] * 2);  // Combine and write to stream
    }
}

When to Use Each

Variant Use When
Simple FIFOs Learning, simple cases, fixed sizes, no external memory
Explicit hls::stream Variable sizes, need depth control, external memory (DDR/HBM)
Stream of blocks Need random access within blocks, block-level synchronization

Key Pragmas

Pragma Purpose
#pragma HLS DATAFLOW Enable concurrent function execution
#pragma HLS INLINE OFF Keep functions as separate hardware modules
#pragma HLS PIPELINE Pipeline the inner loops
#pragma HLS INTERFACE m_axi Memory-mapped AXI interface for DDR access
#pragma HLS PERFORMANCE target_ti=N Target transaction interval
#pragma HLS LOOP_TRIPCOUNT max=N Help tool estimate loop bounds

Performance

With dataflow, functions execute concurrently:

Without DATAFLOW (Sequential):
│funcA│→│funcB│→│funcC│→│funcD│  Total: 4N cycles

With DATAFLOW (Concurrent):
│funcA[0]│funcA[1]│funcA[2]│...
        │funcB[0]│funcB[1]│...
        │funcC[0]│funcC[1]│...
                │funcD[0]│funcD[1]│...  Total: ~N+3 cycles

Speedup: ~4x for this diamond pattern (approaches number of pipeline stages)

Code Generation Guidelines

Kernel Template (Simple FIFOs)

void kernel_name(data_t input[N], data_t output[N]) {
    // Channel declarations (become PIPOs by default)
    data_t c1[N], c2[N], c3[N], c4[N];

#pragma HLS DATAFLOW
    fork_func(input, c1, c2);    // 1 input → 2 outputs
    path1_func(c1, c3);          // Parallel path 1
    path2_func(c2, c4);          // Parallel path 2
    join_func(c3, c4, output);   // 2 inputs → 1 output
}

Kernel Template (Explicit hls::stream)

void kernel_name(vecType* input, vecType* output, int size) {
#pragma HLS INTERFACE m_axi port=input depth=<size>
#pragma HLS INTERFACE m_axi port=output depth=<size>

    hls::stream<vecType> c0, c1, c2, c3, c4, c5;

#pragma HLS DATAFLOW
    load(input, c0, size);           // Memory → Stream
    fork_func(c0, c1, c2, size);     // Fork
    path1_func(c1, c3, size);        // Parallel path 1
    path2_func(c2, c4, size);        // Parallel path 2
    join_func(c3, c4, c5, size);     // Join
    store(c5, output, size);         // Stream → Memory
}

Fork Stage Template

void fork_func(data_t* in, data_t* out1, data_t* out2) {
#pragma HLS INLINE OFF
Loop:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE
        data_t t = in[i];    // Read once
        out1[i] = t;         // Write to path 1
        out2[i] = t;         // Write to path 2
    }
}

Join Stage Template

void join_func(data_t* in1, data_t* in2, data_t* out) {
#pragma HLS INLINE OFF
Loop:
    for (int i = 0; i < N; i++) {
#pragma HLS PIPELINE
        out[i] = combine(in1[i], in2[i]);  // Combine both paths
    }
}

Key Constraints

Constraint Requirement
Dataflow pragma Must have #pragma HLS DATAFLOW in kernel
Function inlining Use #pragma HLS INLINE OFF on all stage functions
Element count Fork outputs and join inputs must have matching counts
Single producer/consumer Each channel must have exactly one writer and one reader

Required Pragmas

Pragma Required? Purpose
#pragma HLS DATAFLOW Yes Enable concurrent function execution
#pragma HLS INLINE OFF Yes Keep functions as separate hardware modules
#pragma HLS PIPELINE Recommended Pipeline inner loops for throughput
#pragma HLS INTERFACE m_axi For DDR access Memory-mapped interface
#pragma HLS PERFORMANCE target_ti=N Optional Target transaction interval

Array Interface Test Bench

#include "diamond.h"
#include <iostream>
#include <fstream>
using namespace std;

int main() {
    data_t test[N];
    data_t outcome[N];
    int retval = 0;
    ofstream FILE;

    // Initialize test vector
    for (int i = 0; i < N; i++) {
        test[i] = (data_t)i;
    }

    FILE.open("result.dat");

    // Execute DUT multiple times
    for (int iter = 0; iter < 3; iter++) {
        diamond(test, outcome);

        for (int i = 0; i < N; i++) {
            FILE << (int)outcome[i] << endl;
        }
    }
    FILE.close();

    // Verify against golden
    retval = system("diff --brief -w result.dat result.golden.dat");
    if (retval != 0) {
        cout << "Test failed!" << endl;
        return 1;
    }
    cout << "Test passed!" << endl;
    return 0;
}

Optimization Guidance for Agents

When to Apply This Pattern

Decision tree:
├── Does the algorithm have natural fork-join structure?
│   ├── Yes → Use diamond pattern
│   └── No → Consider linear pipeline or other topology
├── Are the parallel paths independent?
│   ├── Yes → Diamond is appropriate
│   └── No → May need different synchronization
├── Do paths have similar latency?
│   ├── Yes → Simple arrays (PIPO) work well
│   └── No → Consider explicit streams with depth tuning
└── Need random access within data blocks?
    ├── Yes → Use stream_of_blocks variant
    └── No → Use simple FIFOs or explicit streams

Performance Optimization

Optimization How to Apply Expected Impact
Vectorize data Use hls::vector<T, N> for wider memory bus 16x throughput with 512-bit bus
Separate memory bundles Different bundle= for input/output Enable parallel DDR access
Pipeline all loops Add #pragma HLS PIPELINE to each stage Achieve II=1
Balance path latencies Match iteration counts and processing Prevent stalls

Resource vs Performance Trade-offs

Choice Performance Impact Resource Impact
Simple arrays (PIPO) Automatic double-buffering 2× array size in BRAM
Explicit hls::stream Fine-grained streaming Smaller FIFO buffers
Stream of blocks Random access + streaming Block size × 2 in BRAM
Higher vector width More parallelism More DSPs, wider datapaths

Parameter Tuning

Parameter Low Value High Value Guidance
Array size N Less BRAM More buffering Match data batch size
Vector width Fewer resources Higher throughput Match memory bus width (512-bit = 16×32-bit)
Stream depth Less BRAM More buffering Depth ≥ latency difference between paths
Block size (stream_of_blocks) Fine granularity Coarse granularity Balance access pattern vs overhead

Common Pitfalls

Pitfall Symptom Solution
Missing INLINE OFF Functions merged, no parallelism Add #pragma HLS INLINE OFF to all stage functions
Mismatched element counts Deadlock or incorrect results Ensure fork outputs = join inputs
Multiple readers/writers Vitis HLS error Each channel: exactly 1 producer, 1 consumer
Unbalanced paths One path stalls waiting Balance processing or increase stream depth
Forgetting DATAFLOW pragma Sequential execution Add #pragma HLS DATAFLOW to kernel

Related Patterns

  • Autorestart - Continuous execution with diamond topology
  • Merge-Split - Fan-out/fan-in for N parallel paths
